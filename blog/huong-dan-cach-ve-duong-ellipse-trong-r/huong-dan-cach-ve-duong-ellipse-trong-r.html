library(car)

car:::dataEllipse(x = trees$Height, 
                  y = trees$Volume,
                  xlim = c(60, 95),
                  ylim = c(-10, 80),
                  levels = c(0.5, 0.9, 0.95), 
                  ellipse.label = c(0.5, 0.9, 0.95), 
                  lty = 2, 
                  fill = TRUE, 
                  fill.alpha = 0.1)
